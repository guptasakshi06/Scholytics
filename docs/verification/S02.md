```bash
.venv/bin/python - <<'PY'
import json
from pathlib import Path
from scholarready.dataset import sha256, read_page
rows = {r['sample_id']: r for r in map(json.loads, Path('data/manifests/pilot.jsonl').read_text().splitlines())}
review = json.loads(Path('data/annotations/pilot_review.json').read_text())
assert review['manifest_sha256'] == sha256(Path('data/manifests/pilot.jsonl'))
for item in review['page_content_reviews']:
    r = rows[item['sample_id']]
    assert sha256(Path(r['relative_path'])) == item['artifact_sha256']
    assert sha256(Path(r['annotation_path'])) == item['annotation_sha256']
inventory = json.loads(Path('tests/fixtures/pilot/inventory.json').read_text())
pages = 0
for item in inventory['files']:
    p = Path('tests/fixtures/pilot') / item['file']
    assert sha256(p) == item['sha256'] == sha256(Path(item['original_path']))
    for index in range(1, 3 if item['file'] == 'src-025.pdf' else 2):
        assert read_page(p, index).size == (1200, 1600)
        pages += 1
print(f"PASS: {len(inventory['files'])} fixture files, {pages} pages, {len(review['page_content_reviews'])} current content-review attestations")
PY
```
